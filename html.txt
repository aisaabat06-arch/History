            renderTabs();

        function renderTabs() {
            tabsContainer.innerHTML = '';
            Object.keys(quizData).forEach(category => {
                const button = document.createElement('button');
                button.textContent = category;
                button.className = `whitespace-nowrap px-4 py-2 rounded-full text-sm font-medium transition-colors ${
                    category === currentCategory 
                    ? 'bg-blue-600 text-white shadow-md' 
                    : 'bg-white text-gray-600 hover:bg-gray-100 border border-gray-200'
                }`;
                button.onclick = () => loadCategory(category);
                tabsContainer.appendChild(button);
            });
        }

        function loadCategory(category) {
            currentCategory = category;
            currentQuestions = [...quizData[category]];
            currentCardIndex = 0;
            if (isFlipped) flipCard(false); // Reset flip state without animation delay
            
            renderTabs();
            updateCardDisplay();
            renderList();
        }

        function updateCardDisplay() {
            if (currentQuestions.length === 0) return;
            
            // Allow flip animation to complete before changing text if currently flipped
            if(isFlipped) {
                isFlipped = false;
                flashcardInner.classList.remove('is-flipped');
                setTimeout(() => {
                    setCardContent();
                }, 300);
            } else {
                setCardContent();
            }
        }

        function setCardContent() {
            const card = currentQuestions[currentCardIndex];
            questionEl.textContent = card.q;
            answerEl.textContent = card.a;
            counterEl.textContent = `${currentCardIndex + 1} / ${currentQuestions.length}`;
            currentCategoryTitle.textContent = currentCategory;
        }

        function flipCard(forceState = null) {
            if (forceState !== null) {
                isFlipped = forceState;
            } else {
                isFlipped = !isFlipped;
            }
            
            if (isFlipped) {
                flashcardInner.classList.add('is-flipped');
            } else {
                flashcardInner.classList.remove('is-flipped');
            }
        }

        function nextCard() {
            if (currentCardIndex < currentQuestions.length - 1) {
                currentCardIndex++;
                updateCardDisplay();
            } else {
                // Loop back to start
                currentCardIndex = 0;
                updateCardDisplay();
            }
        }

        function prevCard() {
            if (currentCardIndex > 0) {
                currentCardIndex--;
                updateCardDisplay();
            } else {
                // Go to end
                currentCardIndex = currentQuestions.length - 1;
                updateCardDisplay();
            }
        }

        function shuffleCards() {
            for (let i = currentQuestions.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [currentQuestions[i], currentQuestions[j]] = [currentQuestions[j], currentQuestions[i]];
            }
            currentCardIndex = 0;
            updateCardDisplay();
        }

                    currentQuestions = [...quizData[currentCategory]];
                    currentCardIndex = index;
                    updateCardDisplay();

                        <div class="flex-1">
                            <p class="font-medium text-gray-800">${item.q}</p>
                            <p class="text-green-600 font-semibold mt-1">— ${item.a}</p>
                        </div>
                    </div>
                `;
                questionsList.appendChild(li);
            });
        }

                flipCard();
            }
        });
